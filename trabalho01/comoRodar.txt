# Como executar o projeto

# 1. Entre na pasta do projeto.
cd trabalho01

# 2. Crie o ambiente virtual dentro da pasta do projeto.
python -m venv venv

# 3. Ative o ambiente virtual.
# CMD:
venv\Scripts\activate
# PowerShell:
.\venv\Scripts\Activate.ps1

# 4. Instale as dependências do projeto.
python -m pip install --upgrade pip
pip install -r requirements.txt

# 5. Atualize o banco local (inclui pedidos, itens e dados de livros).
python manage.py migrate

# 6. Inicie o servidor do projeto.
python manage.py runserver

# Alternativa no Windows:
# Dê duplo clique em iniciar.bat. Ele cria o ambiente virtual,
